```back/library/db-init/02-dummydata.sql
CREATE PROCEDURE InsertDummyBorrowRecords(IN record_count INT)
BEGIN
    DECLARE i INT DEFAULT 0;
    DECLARE rand_book_id VARCHAR(255);
    DECLARE rand_user_id INT;
    DECLARE user_id INT;
    DECLARE rand_borrow_at DATETIME;
    DECLARE rand_return_at DATETIME;

    SET user_id = 3;

    WHILE i < record_count DO
        -- 랜덤한 book_id 가져오기
        SELECT book_id INTO rand_book_id
        FROM book
        ORDER BY RAND()
        LIMIT 1;

        SET rand_borrow_at = DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 30) DAY);

        IF RAND() > 0.5 THEN
            SET rand_return_at = DATE_ADD(rand_borrow_at, INTERVAL FLOOR(RAND() * 14) DAY);
        ELSE
            SET rand_return_at = NULL;
        END IF;

        INSERT INTO borrow_record (borrow_at, return_at, book_id, user_id)
        VALUES (rand_borrow_at, rand_return_at, rand_book_id, rand_user_id);

        SET i = i + 1;
    END WHILE;
END $$

DELIMITER ;
```

```back/library/db-init/03-mostprocedure.sql
CREATE TABLE book_rank (
    rank_id INT PRIMARY KEY AUTO_INCREMENT,
    borrow_count INT NOT NULL,
    ranking_period TINYINT NOT NULL,
    updated_at DATE NOT NULL,
    book_id varchar(255) NOT NULL,
    UNIQUE (book_id, ranking_period)
);

ALTER TABLE book_rank
    ADD CONSTRAINT
        FOREIGN KEY (book_id) REFERENCES `book` (book_id);

DELIMITER $$

CREATE PROCEDURE get_weekly_top_books()
BEGIN
  REPLACE INTO book_rank (book_id, borrow_count, ranking_period, updated_at)
  SELECT b.book_id, COUNT(b.book_id) AS borrow_count, 0, NOW()
  FROM book b
  JOIN borrow_record br ON b.book_id = br.book_id
  WHERE br.borrow_at BETWEEN
  DATE_SUB(DATE(NOW()), INTERVAL WEEKDAY(NOW()) + 7 DAY) AND
  DATE_SUB(DATE(NOW()), INTERVAL WEEKDAY(NOW()) + 1 DAY)
  GROUP BY b.book_id
  ORDER BY borrow_count DESC
  LIMIT 10;
END $$

CREATE PROCEDURE get_monthly_top_books()
BEGIN
  REPLACE INTO book_rank (book_id, borrow_count, ranking_period, updated_at)
  SELECT b.book_id, COUNT(b.book_id) AS borrow_count, 1, NOW()
  FROM book b
  JOIN borrow_record br ON b.book_id = br.book_id
  WHERE br.borrow_at BETWEEN 
  DATE_FORMAT(DATE_SUB(NOW(), INTERVAL 1 MONTH), '%Y-%m-01')
  AND LAST_DAY(DATE_SUB(NOW(), INTERVAL 1 MONTH))
  GROUP BY b.book_id
  ORDER BY borrow_count DESC
  LIMIT 10;
END $$

DELIMITER ;

CALL get_weekly_top_books();
CALL get_monthly_top_books();
```

```back/library/src/main/java/com/example/library/config/MqttConfig.java
package com.example.library.config;

import org.eclipse.paho.client.mqttv3.MqttConnectOptions;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.integration.annotation.ServiceActivator;
import org.springframework.integration.channel.DirectChannel;
import org.springframework.integration.mqtt.core.DefaultMqttPahoClientFactory;
import org.springframework.integration.mqtt.core.MqttPahoClientFactory;
import org.springframework.integration.mqtt.inbound.MqttPahoMessageDrivenChannelAdapter;
import org.springframework.integration.mqtt.outbound.MqttPahoMessageHandler;
import org.springframework.integration.mqtt.support.DefaultPahoMessageConverter;
import org.springframework.messaging.MessageChannel;
import org.springframework.messaging.MessageHandler;

@Configuration
public class MqttConfig {

    private final String MQTT_BROKER_URL = "tcp://mosquitto:1883";  // MQTT 브로커 주소
    private final String MQTT_CLIENT_ID = "spring-boot-server";  // 클라이언트 ID

    @Bean
    public MqttPahoClientFactory mqttClientFactory() {
        DefaultMqttPahoClientFactory factory = new DefaultMqttPahoClientFactory();
        MqttConnectOptions options = new MqttConnectOptions();
        options.setServerURIs(new String[]{MQTT_BROKER_URL});
        options.setCleanSession(true);
        factory.setConnectionOptions(options);
        return factory;
    }

    // ✅ 1. MQTT 메시지를 발행(Publish)하는 Bean
    @Bean
    @ServiceActivator(inputChannel = "mqttOutboundChannel")
    public MessageHandler mqttOutbound() {
        MqttPahoMessageHandler messageHandler =
            new MqttPahoMessageHandler(MQTT_CLIENT_ID, mqttClientFactory());
        messageHandler.setAsync(true);
        return messageHandler;
    }

    @Bean
    public MessageChannel mqttOutboundChannel() {
        return new DirectChannel();
    }

    // ✅ 2. MQTT 메시지를 구독(Subscribe)하는 Bean
    @Bean
    public MessageChannel mqttInboundChannel() {
        return new DirectChannel();
    }

    @Bean
    public MqttPahoMessageDrivenChannelAdapter mqttInbound() {
        MqttPahoMessageDrivenChannelAdapter adapter =
            new MqttPahoMessageDrivenChannelAdapter(MQTT_CLIENT_ID + "_sub",
                mqttClientFactory(), "topic/book/borrow", "topic/user/borrow"
                , "topic/book/return/start", "topic/book/return/finish",
                "topic/log/save"); // 여러 개의 주제 구독 가능
        adapter.setCompletionTimeout(5000);
        adapter.setConverter(new DefaultPahoMessageConverter());
        adapter.setQos(1);
        adapter.setOutputChannel(mqttInboundChannel());
        return adapter;
    }
}
```

```back/library/src/main/java/com/example/library/config/RedisConfig.java
package com.example.library.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import org.springframework.data.redis.connection.RedisConnectionFactory;
import org.springframework.data.redis.connection.lettuce.LettuceConnectionFactory;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.data.redis.serializer.StringRedisSerializer;

@Configuration
public class RedisConfig {

    private String redisHost = "redis-container";

    private int redisPort = 6379;

    @Bean
    public RedisConnectionFactory redisConnectionFactory() {
        return new LettuceConnectionFactory(redisHost, redisPort);
    }

    @Bean
    @Primary
    public RedisTemplate<String, String> redisTemplate() {
        RedisTemplate<String, String> template = new RedisTemplate<>();
        template.setConnectionFactory(redisConnectionFactory());
        template.setKeySerializer(new StringRedisSerializer());
        template.setValueSerializer(new StringRedisSerializer());
        return template;
    }
}
```

```back/library/src/main/java/com/example/library/domain/arrivezone/controller/ArriveZoneController.java
import com.example.library.domain.arrivezone.dto.request.ArriveZoneResetRequestDto;
```

```back/library/src/main/java/com/example/library/domain/arrivezone/service/ArriveZoneService.java
import com.example.library.domain.arrivezone.dto.response.ArriveZoneResponseDto.ArriveZoneDto;
import com.example.library.domain.arrivezone.entity.ArriveZone;
import com.example.library.domain.arrivezone.repository.ArriveZoneRepository;
import jakarta.transaction.Transactional;
import java.util.NoSuchElementException;
import org.springframework.stereotype.Service;

@Service
@Transactional
public class ArriveZoneService {

    private ArriveZoneRepository arriveZoneRepository;

    public ArriveZoneService(ArriveZoneRepository arriveZoneRepository) {
        this.arriveZoneRepository = arriveZoneRepository;
    }

    public ArriveZoneResponseDto getArriveZoneList() {
        return new ArriveZoneResponseDto(arriveZoneRepository.findAll().stream()
            .map(this::changeToDto)
            .toList());
    }

    private ArriveZoneDto changeToDto(ArriveZone arriveZone) {
        return new ArriveZoneDto(
            arriveZone.getArriveZoneId(),
            arriveZone.getArriveZoneName(),
            arriveZone.getArriveZoneBook(),
            arriveZone.getArriveZoneTime()
        );
    }

    public void resetArriveZone(ArriveZoneResetRequestDto arriveZoneResetRequestDto) {
        ArriveZone targetArriveZone = arriveZoneRepository
            .findById(arriveZoneResetRequestDto.getArriveZoneId())
            .orElseThrow(() -> new NoSuchElementException("해당 회수 구역의 데이터가 없습니다."));

        targetArriveZone.resetBookCount();
    }
}
```

```back/library/src/main/java/com/example/library/domain/book/entity/Book.java
package com.example.library.domain.book.entity;

import com.example.library.domain.arrivezone.entity.ArriveZone;
import jakarta.persistence.CascadeType;
import jakarta.persistence.Entity;
import jakarta.persistence.FetchType;
import jakarta.persistence.Id;
import jakarta.persistence.JoinColumn;
import jakarta.persistence.ManyToOne;
import jakarta.persistence.OneToMany;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;
import java.util.NoSuchElementException;

@Entity
public class Book {

    @Id
    private String bookId;
    private String bookIsbn;
    private String bookTitle;
    private String bookWriter;
    private String bookPublisher;
    private int bookStatus;
    private String bookSign;
    private LocalDateTime bookCreatedAt;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "arrive_zone_id")
    private ArriveZone arriveZone;

    @OneToMany(mappedBy = "book", cascade = CascadeType.REMOVE)
    private List<BorrowRecord> borrowRecordList = new ArrayList<>();

    public Book() {
    }

    public Book(String bookId, String bookIsbn, String bookTitle, String bookWriter,
        String bookPublisher, int bookStatus, String bookSign, LocalDateTime bookCreatedAt,
        ArriveZone arriveZone) {
        this.bookId = bookId;
        this.bookIsbn = bookIsbn;
        this.bookTitle = bookTitle;
        this.bookWriter = bookWriter;
        this.bookPublisher = bookPublisher;
        this.bookStatus = bookStatus;
        this.bookSign = bookSign;
        this.bookCreatedAt = bookCreatedAt;

        this.arriveZone = arriveZone;
        arriveZone.addBook(this);
    }

    public String getBookId() {
        return bookId;
    }

    public void setBookId(String bookId) {
        this.bookId = bookId;
    }

    public String getBookIsbn() {
        return bookIsbn;
    }

    public void setBookIsbn(String bookIsbn) {
        this.bookIsbn = bookIsbn;
    }

    public String getBookTitle() {
        return bookTitle;
    }

    public void setBookTitle(String bookTitle) {
        this.bookTitle = bookTitle;
    }

    public String getBookWriter() {
        return bookWriter;
    }

    public void setBookWriter(String bookWriter) {
        this.bookWriter = bookWriter;
    }

    public String getBookPublisher() {
        return bookPublisher;
    }

    public void setBookPublisher(String bookPublisher) {
        this.bookPublisher = bookPublisher;
    }

    public int getBookStatus() {
        return bookStatus;
    }

    public void setBookStatus(int bookStatus) {
        this.bookStatus = bookStatus;
    }

    public String getBookSign() {
        return bookSign;
    }

    public void setBookSign(String bookSign) {
        this.bookSign = bookSign;
    }

    public LocalDateTime getBookCreatedAt() {
        return bookCreatedAt;
    }

    public void setBookCreatedAt(LocalDateTime bookCreatedAt) {
        this.bookCreatedAt = bookCreatedAt;
    }

    public ArriveZone getArriveZone() {
        return arriveZone;
    }

    public void setArriveZone(ArriveZone arriveZone) {
        this.arriveZone.deleteBook(this);
        arriveZone.addBook(this);

        this.arriveZone = arriveZone;
    }

    public List<BorrowRecord> getBorrowRecordList() {
        return borrowRecordList;
    }

    public void addBorrowRecord(BorrowRecord borrowRecord) {
        if (borrowRecordList.contains(borrowRecord)) {
            return;
        }

        borrowRecordList.add(borrowRecord);
    }

    public LocalDate getBookReturn() {
        if (borrowRecordList.size() == 0 || bookStatus != 0) {
            return null;
        }

        return borrowRecordList.get(borrowRecordList.size() - 1)
            .getBorrowAt()
            .plusWeeks(2)
            .toLocalDate();
    }

    public BorrowRecord getLastBorrowRecord() {
        if (borrowRecordList.size() == 0 || bookStatus != 0) {
            throw new NoSuchElementException("대출 기록이 없습니다");
        }

        return borrowRecordList.get(borrowRecordList.size() - 1);
    }
}
```

```back/library/src/main/java/com/example/library/domain/book/service/BookConnectService.java
import java.util.NoSuchElementException;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Service;

@Service
public class BookConnectService {

    private final RedisTemplate<String, String> redisTemplate;

    private BookRepository bookRepository;
    private RobotRepository robotRepository;

    private final static String REDIS_KEY = "book_borrow/";

    public BookConnectService(RedisTemplate<String, String> redisTemplate,
        BookRepository bookRepository, RobotRepository robotRepository) {
        this.redisTemplate = redisTemplate;
        this.bookRepository = bookRepository;
        this.robotRepository = robotRepository;
    }

    public void saveBookInfo(BookRobotRequestDto bookRobotRequestDto) {
        if (!robotRepository.existsById(bookRobotRequestDto.getRobotId())) {
            throw new NoSuchElementException("로봇 정보가 존재하지 않습니다!");
        }

        if (!bookRepository.existsById(bookRobotRequestDto.getBookId())) {
            throw new NoSuchElementException("책 정보가 존재하지 않습니다!");
        }

        redisTemplate.opsForValue()
            .set(REDIS_KEY + bookRobotRequestDto.getRobotId(),
                bookRobotRequestDto.getBookId(), Duration.ofMinutes(5));
    }

    public BookFrontResponseDto getBookInfo(BookFrontRequestDto bookFrontRequestDto) {
        String targetRedisKey = REDIS_KEY + bookFrontRequestDto.getRobotId();
        String targetBookId = redisTemplate.opsForValue().get(targetRedisKey);

        if (targetBookId == null) {
            throw new NoSuchElementException("저장된 책 데이터가 없습니다.");
        }

        Book targetBook = bookRepository.findById(targetBookId)
            .orElseThrow(() -> new NoSuchElementException("해당 바코드 데이터가 존재하지 않습니다."));

        redisTemplate.delete(targetRedisKey);
        return new BookFrontResponseDto(
            targetBook.getBookId(),
            targetBook.getBookIsbn(),
            BookStatus.values()[targetBook.getBookStatus()].getStatusDisc()
        );
    }
```

```back/library/src/main/java/com/example/library/domain/book/service/BookRobotService.java
        User targetUser = userRepository.findById(bookBorrowRequestDto.getUserId())
            .orElseThrow(() -> new NoSuchElementException("해당 유저 데이터가 존재하지 않습니다."));

        Book targetBook = bookRepository.findById(bookBorrowRequestDto.getBookId())
            .orElseThrow(() -> new NoSuchElementException("해당 책 데이터가 존재하지 않습니다."));

            throw new NoSuchElementException("이미 대출 중인 책입니다.");

        Robot targetRobot = robotRepository.findById(bookBorrowRequestDto.getRobotId())
            .orElseThrow(() -> new NoSuchElementException("해당 로봇 데이터가 존재하지 않습니다."));

        targetBook.setBookStatus(0);
        borrowRecordRepository.save(newBorrowRecord);
        robotLogRepository.save(newRobotLog);
        noticeRepository.save(newNotice);
```

```back/library/src/main/java/com/example/library/domain/code/service/CodeService.java
package com.example.library.domain.code.service;

import com.example.library.domain.code.dto.request.CodeRequestDto;
import com.example.library.domain.user.entity.User;
import com.example.library.domain.user.repository.UserRepository;
import com.google.zxing.BarcodeFormat;
import com.google.zxing.MultiFormatWriter;
import com.google.zxing.client.j2se.MatrixToImageWriter;
import com.google.zxing.common.BitMatrix;
import jakarta.transaction.Transactional;
import java.io.ByteArrayOutputStream;
import java.util.NoSuchElementException;
import org.springframework.stereotype.Service;

@Service
@Transactional
public class CodeService {

    UserRepository userRepository;

    public CodeService(UserRepository userRepository) {
        this.userRepository = userRepository;
    }

    public byte[] getCode(CodeRequestDto codeRequestDto) {
        int width = 200;
        int height = 200;

        User targetUser = userRepository.findById(codeRequestDto.getUserId())
            .orElseThrow(() -> new NoSuchElementException("해당 유저가 존재하지 않습니다."));

        // QR Code - Image 생성. : 1회성으로 생성해야 하기 때문에
        // stream으로 Generate(1회성이 아니면 File로 작성 가능.)
        try {
            BitMatrix encode = new MultiFormatWriter()
                .encode(targetUser.getUserCode(), BarcodeFormat.QR_CODE, width, height);
            //output Stream
            ByteArrayOutputStream out = new ByteArrayOutputStream();

            //Bitmatrix, file.format, outputStream
            MatrixToImageWriter.writeToStream(encode, "PNG", out);

            return out.toByteArray();

        } catch (Exception e) {
            throw new NoSuchElementException("QR 코드 작성 과정에서 오류!");
        }
    }
}
```

```back/library/src/main/java/com/example/library/domain/mqtt/MqttHandler.java
import com.example.library.domain.mqtt.controller.MqttController;

import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.integration.annotation.ServiceActivator;
import org.springframework.messaging.Message;
import org.springframework.stereotype.Component;

@Component
public class MqttHandler {

    private ObjectMapper objectMapper = new ObjectMapper();

    private MqttController mqttController;

    public MqttHandler(MqttController mqttController) {
        this.mqttController = mqttController;
    }

    @ServiceActivator(inputChannel = "mqttInboundChannel")
    public void handleMqttMessage(Message<String> message) {
        String topic = message.getHeaders().get("mqtt_receivedTopic", String.class);
        String payload = message.getPayload();

        try {
            switch (topic) {
                case "topic/book/borrow" -> {
                    BookSaveRequestDto bookSaveRequestDto = objectMapper.readValue(
                        payload, BookSaveRequestDto.class);
                    mqttController.saveBook(bookSaveRequestDto);
                }
                case "topic/user/borrow" -> {
                    UserSaveRequestDto userSaveRequestDto = objectMapper.readValue(
                        payload, UserSaveRequestDto.class);
                    mqttController.saveUser(userSaveRequestDto);
                }
                case "topic/book/return/start" -> {
                    BookReturnDepartRequestDto bookReturnDepartRequestDto = objectMapper.readValue(
                        payload, BookReturnDepartRequestDto.class);
                    mqttController.bookReturnStart(bookReturnDepartRequestDto);
                }
                case "topic/book/return/finish" -> {
                    BookReturnFinishRequestDto bookReturnFinishRequestDto = objectMapper.readValue(
                        payload, BookReturnFinishRequestDto.class);
                    mqttController.bookReturnFinish(bookReturnFinishRequestDto);
                }
                case "topic/log/save" -> {
                    RobotLogSaveRequestDto robotLogSaveRequestDto = objectMapper.readValue(
                        payload, RobotLogSaveRequestDto.class);
                    mqttController.saveRobotLog(robotLogSaveRequestDto);
                }
                default -> mqttController.sendErrorMessage(payload, "등록된 topic이 아니에요.");
            }
        } catch (Exception e) {
            mqttController.sendErrorMessage(payload, "JSON 파싱 에러!");
        }
    }
}
```

```back/library/src/main/java/com/example/library/domain/otp/service/OtpService.java
package com.example.library.domain.otp.service;

import com.example.library.domain.otp.dto.request.SendOtpRequestDto;
import com.example.library.domain.otp.dto.request.VerifyOtpRequestDto;
import java.time.Duration;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Service;

@Service
public class OtpService {

    private final RedisTemplate<String, String> redisTemplate;

    public OtpService(RedisTemplate<String, String> redisTemplate) {
        this.redisTemplate = redisTemplate;
    }

    // OTP 저장 (5분 TTL)
    public void saveOtp(SendOtpRequestDto sendOtpRequestDto, String otp) {
        redisTemplate.opsForValue()
            .set(sendOtpRequestDto.getUserEmail(), otp, Duration.ofMinutes(5));
    }

    //OTP 검증

    public boolean validateOtp(VerifyOtpRequestDto verifyOtpRequestDto) {
        String storedOtp = redisTemplate.opsForValue().get(verifyOtpRequestDto.getUserEmail());
        if (storedOtp != null && storedOtp.equals(verifyOtpRequestDto.getCertNumber())) {
            return true;
        }
        return false;
    }
}
```

```back/library/src/main/java/com/example/library/domain/robot/service/RobotService.java
            return false;
```

```back/library/src/main/java/com/example/library/domain/token/service/AccessTokenService.java
package com.example.library.domain.token.service;

import com.example.library.domain.token.exception.AuthorizationException;
import com.example.library.domain.token.exception.TokenException;
import com.example.library.util.EncryptKeys;
import com.nimbusds.jose.EncryptionMethod;
import com.nimbusds.jose.JOSEException;
import com.nimbusds.jose.JWEAlgorithm;
import com.nimbusds.jose.JWEHeader;
import com.nimbusds.jose.JWEObject;
import com.nimbusds.jose.Payload;
import com.nimbusds.jose.crypto.RSADecrypter;
import com.nimbusds.jose.crypto.RSAEncrypter;
import com.nimbusds.jwt.JWTClaimsSet;
import jakarta.servlet.http.HttpServletRequest;
import java.security.PrivateKey;
import java.text.ParseException;
import java.time.Duration;
import java.util.Date;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Service;

@Service
public class AccessTokenService {

    private static final String AT_BLACK_LIST_PREFIX = "access_token_";

    private final RedisTemplate<String, String> redisTemplate;

    public AccessTokenService(RedisTemplate<String, String> redisTemplate) {
        this.redisTemplate = redisTemplate;
    }

    public String generateAccessToken(int userId) {
        // JWT Claims
        JWTClaimsSet claims = new JWTClaimsSet.Builder()
            .claim("userId", userId)
            .expirationTime(new Date(System.currentTimeMillis() + 30 * 60 * 1000))
            .build();

        // JWE Header
        JWEHeader header = new JWEHeader.Builder(JWEAlgorithm.RSA_OAEP_256,
            EncryptionMethod.A128GCM)
            .contentType("JWT")
            .build();

        // JWE Object
        JWEObject jweObject = new JWEObject(header, new Payload(claims.toJSONObject()));

        // Encrypt JWE with RSA Public Key
        try {
            jweObject.encrypt(new RSAEncrypter(EncryptKeys.getPublicKey()));
        } catch (JOSEException e) {
            throw new TokenException("access token 암호화 과정에서 문제!");
        }

        redisTemplate.delete(AT_BLACK_LIST_PREFIX + userId);
        return jweObject.serialize();
    }

    // JWE 검증 메서드
    public JWTClaimsSet decryptJWE(String jweToken) throws ParseException, JOSEException {
        // 1. JWE 토큰 파싱
        JWEObject jweObject = JWEObject.parse(jweToken);

        // 2. RSA 개인 키를 사용해 복호화
        PrivateKey privateKey = EncryptKeys.getPrivateKey();
        jweObject.decrypt(new RSADecrypter(privateKey));

        // 3. Payload에서 JWT Claims 추출
        String payload = jweObject.getPayload().toString();

        // 4. Payload를 JWTClaimsSet으로 변환
        return JWTClaimsSet.parse(payload);
    }

    // Payload 검증 메서드
    public boolean isValidClaims(JWTClaimsSet claims) {
        try {
            // 1. 만료 시간 (exp) 검증
            if (claims.getExpirationTime() == null || claims.getExpirationTime()
                .before(new Date())) {
                return false;
            }

            int userId = ((Long) claims.getClaim("userId")).intValue();

            return !redisTemplate.hasKey(AT_BLACK_LIST_PREFIX + userId);

            // 추가적인 검증 로직 필요시 여기에 추가
        } catch (Exception e) {
            throw new AuthorizationException("유효하지 않은 Access Token");
        }
    }

    public boolean isValidAuthClaims(JWTClaimsSet claims) {
        if (!isValidClaims(claims)) {
            return false;
        }

        try {
            int userId = ((Long) claims.getClaim("userId")).intValue();

            return userId < 6;
        } catch (Exception e) {
            throw new AuthorizationException("유효하지 않은 Access Token");
        }
    }

    public String resolveAccessToken(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        if (bearerToken != null && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7); // "Bearer " 제거
        }
        return null;
    }

    public int getUserId(HttpServletRequest request) {
        try {
            return ((Long) decryptJWE(resolveAccessToken(request)).getClaim("userId")).intValue();
        } catch (Exception e) {
            throw new AuthorizationException("유효하지 않은 Access Token");
        }
    }

    public void addBlackList(int userId) {
        redisTemplate.opsForValue()
            .set(AT_BLACK_LIST_PREFIX + userId, "1", Duration.ofMinutes(30));
    }
}
```

```back/library/src/main/java/com/example/library/domain/token/service/RefreshTokenService.java
package com.example.library.domain.token.service;

import com.example.library.util.EncryptKeys;
import com.example.library.util.OpenCrypt;
import jakarta.servlet.http.Cookie;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.time.Duration;
import java.util.UUID;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Service;

@Service
public class RefreshTokenService {

    private static final String RT_PREFIX = "refresh_token_";

    private final RedisTemplate<String, String> redisTemplate;

    public RefreshTokenService(RedisTemplate<String, String> redisTemplate) {
        this.redisTemplate = redisTemplate;
    }

    public void insertRefreshToken(UUID refreshToken, int userId) {

        String redisRefreshToken = encryptRefreshToken(refreshToken);

        redisTemplate.opsForValue()
            .set(RT_PREFIX + userId, redisRefreshToken, Duration.ofDays(30));
    }


    public boolean validateRefreshToken(HttpServletRequest request, int userId) {
        String storedRefreshToken = redisTemplate.opsForValue().get(RT_PREFIX + userId);
        UUID refreshToken = resolveRefreshToken(request);

        if (storedRefreshToken == null || refreshToken == null) {
            return false;
        }

        String requestRefreshToken = encryptRefreshToken(refreshToken);

        return storedRefreshToken.equals(requestRefreshToken);
    }

    private String encryptRefreshToken(UUID refreshToken) {
        return OpenCrypt.byteArrayToHex(
            OpenCrypt.getSHA256(refreshToken.toString(), EncryptKeys.getRefreshTokenKey()));
    }

    private UUID resolveRefreshToken(HttpServletRequest request) {
        // Refresh Token은 HttpOnly Cookie로 전달되므로 쿠키에서 추출
        if (request.getCookies() != null) {
            for (Cookie cookie : request.getCookies()) {
                if ("refreshToken".equals(cookie.getName())) {
                    return UUID.fromString(cookie.getValue());
                }
            }
        }
        return null;
    }

    public void addRefreshTokenToCookie(HttpServletResponse response, UUID newRefreshToken) {
        // HttpOnly Cookie로 refresh token 설정
        Cookie cookie = new Cookie("refreshToken", newRefreshToken.toString());
        cookie.setHttpOnly(true);  // HttpOnly 설정
        cookie.setSecure(true);     // HTTPS 프로토콜을 사용할 때는 Secure 설정
        cookie.setPath("/");       // 경로 설정
        cookie.setMaxAge(60 * 60 * 24 * 30);  // 만료 시간 (예: 30일)// 크로스 도메인 허용
        cookie.setAttribute("SameSite", "None");
        response.addCookie(cookie);
    }

    public void deleteRefreshToken(int userId) {
        redisTemplate.delete(RT_PREFIX + userId);
    }
}
```

```back/library/src/main/java/com/example/library/domain/user/service/UserAuthService.java
            .orElseThrow(() -> new NoSuchElementException("일치하는 정보를 갖는 유저가 존재하지 않습니다."));

            throw new NoSuchElementException("일치하는 정보를 갖는 유저가 존재하지 않습니다.");

        Salt targetSalt = saltRepository.findById(targetUser.getUserId())
            .orElseThrow(() -> new NoSuchElementException("데이터 베이스 오류!"));

        String newPassword = PasswordGenerator.generatePassword(12);
        String hashedPassword = OpenCrypt.byteArrayToHex(
            OpenCrypt.getSHA256(newPassword, targetSalt.getSalt()));

        targetUser.setUserPassword(hashedPassword);
        emailService.sendUserPw(targetUser.getUserEmail(), newPassword);
        return newPassword;
    }

    public UserLoginResponseDto userLogin(UserLoginRequestDto userLoginRequestDto,

            .orElseThrow(() -> new NoSuchElementException("아이디 또는 비밀번호가 잘못되었습니다."));

        if (targetUser.getUserId() < 6) {
            if (!targetUser.getUserPassword().equals(userLoginRequestDto.getUserPassword())) {
                throw new NoSuchElementException("아이디 또는 비밀번호가 일치하지 않습니다.");
            }

            tokenService.setTokens(targetUser.getUserId(), response);

            return new UserLoginResponseDto(targetUser.getUserId(), targetUser.getUserName());
        }

        Salt targetSalt = saltRepository.findById(targetUser.getUserId())
            .orElseThrow(() -> new NoSuchElementException("데이터 베이스 오류!"));

        String hashedPassword = OpenCrypt.byteArrayToHex(
            OpenCrypt.getSHA256(userLoginRequestDto.getUserPassword(), targetSalt.getSalt()));

        if (!hashedPassword.equals(targetUser.getUserPassword())) {
            throw new NoSuchElementException("아이디 또는 비밀번호가 일치하지 않습니다.");
        }

        tokenService.setTokens(targetUser.getUserId(), response);

            if (!targetPassword.equals(targetUser.getUserPassword())) {
                throw new NoSuchElementException("비밀번호가 일치하지 않습니다.");
            }
        }

        }
```

```back/library/src/main/java/com/example/library/domain/user/service/UserConnectService.java
        this.redisTemplate = redisTemplate;
        this.userRepository = userRepository;
        this.robotRepository = robotRepository;

            .set(REDIS_KEY + userRobotRequestDto.getRobotId(),
                userRobotRequestDto.getUserCode(), Duration.ofMinutes(5));
```

```back/library/src/main/java/com/example/library/util/OpenCrypt.java
package com.example.library.util;

import com.example.library.domain.user.dto.PasswordEncryptionResultDto;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;

public class OpenCrypt {

    public static PasswordEncryptionResultDto encryptPw(String source) {

        SecureRandom sr = new SecureRandom();

        byte[] saltbyte = new byte[20];

        sr.nextBytes(saltbyte);

        StringBuffer sb = new StringBuffer();
        for (byte b : saltbyte) {
            sb.append(String.format("%02x", b));
        }

        String salt = sb.toString();

        byte[] hashedPw = getSHA256(source, salt);
        String hashedPassword = byteArrayToHex(hashedPw);

        return new PasswordEncryptionResultDto(hashedPassword, salt);
    }

    public static byte[] getSHA256(String source, String salt) {
        byte[] byteData = null;
        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            md.update(source.getBytes());
            md.update(salt.getBytes());
            byteData = md.digest();
            System.out.println(
                "원문: " + source + "   SHA-256: " + byteData.length + "," + byteArrayToHex(
                    byteData));
        } catch (NoSuchAlgorithmException e) {
            e.printStackTrace();
        }
        return byteData;
    }

    public static String byteArrayToHex(byte[] ba) {
        if (ba == null || ba.length == 0) {
            return null;
        }

        StringBuffer sb = new StringBuffer(ba.length * 2);
        String hexNumber;
        for (int x = 0; x < ba.length; x++) {
            hexNumber = "0" + Integer.toHexString(0xff & ba[x]);

            sb.append(hexNumber.substring(hexNumber.length() - 2));
        }
        return sb.toString();
    }

}
```

```back/library/src/main/java/com/example/library/util/PasswordGenerator.java
package com.example.library.util;

import java.security.SecureRandom;
import java.util.Random;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class PasswordGenerator {

    private static final String UPPER = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    private static final String LOWER = "abcdefghijklmnopqrstuvwxyz";
    private static final String DIGITS = "0123456789";
    private static final String SPECIAL = "!@#$%^&*()-_=+<>?";
    private static final String ALL_CHARS = UPPER + LOWER + DIGITS + SPECIAL;

    private static final Random random = new SecureRandom(); // 보안적으로 안전한 난수 생성기

    public static String generatePassword(int length) {
        // 필수 요소 하나씩 포함
        String password = String.valueOf(UPPER.charAt(random.nextInt(UPPER.length()))) +
            LOWER.charAt(random.nextInt(LOWER.length())) +
            DIGITS.charAt(random.nextInt(DIGITS.length())) +
            SPECIAL.charAt(random.nextInt(SPECIAL.length()));

        // 나머지 길이만큼 무작위 문자 추가
        password += IntStream.range(0, length - 4)
            .mapToObj(i -> String.valueOf(ALL_CHARS.charAt(random.nextInt(ALL_CHARS.length()))))
            .collect(Collectors.joining());

        // 무작위로 섞기
        return shuffleString(password);
    }

    private static String shuffleString(String input) {
        char[] array = input.toCharArray();
        for (int i = array.length - 1; i > 0; i--) {
            int j = random.nextInt(i + 1);
            char temp = array[i];
            array[i] = array[j];
            array[j] = temp;
        }
        return new String(array);
    }
}
```
