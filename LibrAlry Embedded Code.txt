# image crop
def autocrop(image, threshold=0):
    """Crops any edges below or equal to threshold
    Crops blank image to 1x1.
    Returns cropped image.
    """
    if len(image.shape) == 3:
        flatImage = np.max(image, 2)
    else:
        flatImage = image
    assert len(flatImage.shape) == 2

    rows = np.where(np.max(flatImage, 0) > threshold)[0]
    if rows.size:
        cols = np.where(np.max(flatImage, 1) > threshold)[0]
        image = image[cols[0]: cols[-1] + 1, rows[0]: rows[-1] + 1]
    else:
        image = image[:1, :1]

    return image

        print(f"[ERROR] WebSocket send error: {e}")

async def scan_qr_code():
    async with websockets.connect(SERVER_URL) as websocket:
        while True:
            # Capture frame from the camera
            frame = picam2.capture_array()
            frame = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)  # Convert from RGB to BGR for OpenCV
            frame = imutils.resize(frame, width=640)

            # Crop function
            crop = autocrop(frame, 50)

            # Find barcodes in the frame and decode them
            barcodes = pyzbar.decode(crop)

            for barcode in barcodes:
                # Draw red box
                (x, y, w, h) = barcode.rect
                cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 0, 255), 2)

                # Byte to text
                barcodeData = barcode.data.decode("utf-8")
                barcodeType = barcode.type

                # Draw data and type on the image
                text = "{} ({})".format(barcodeData, barcodeType)

        # high = self.bus.read_byte_data(self.address, register)

        # value = (high << 8) + low
        # if value >= 0x8000:
        #     return -((65535 - value) + 1)
        # else:
        #     return value
        if self.bus is None:
            raise RuntimeError("MPU6050 not initialized")  # 예외 발생
        try:
            high = self.bus.read_byte_data(self.address, register)
            low = self.bus.read_byte_data(self.address, register + 1)

                return value
        except OSError as e:
            rospy.logwarn(f"I2C Read Error at register 0x{register:02X}: {e}")
            raise  # 예외를 다시 발생시켜 상위 코드에서 처리하도록 함


    def get_accel_data(self):
        x = self.read_i2c_word(self.ACCEL_XOUT0) / self.ACCEL_SCALE_MODIFIER_2G * self.GRAVITY_MS2
        y = self.read_i2c_word(self.ACCEL_XOUT0 + 2) / self.ACCEL_SCALE_MODIFIER_2G * self.GRAVITY_MS2
        z = self.read_i2c_word(self.ACCEL_XOUT0 + 4) / self.ACCEL_SCALE_MODIFIER_2G * self.GRAVITY_MS2
        return {'x': x, 'y': y, 'z': z}

    def get_gyro_data(self):
        x = self.read_i2c_word(self.GYRO_XOUT0) / self.GYRO_SCALE_MODIFIER_250DEG
        y = self.read_i2c_word(self.GYRO_XOUT0 + 2) / self.GYRO_SCALE_MODIFIER_250DEG
        z = self.read_i2c_word(self.GYRO_XOUT0 + 4) / self.GYRO_SCALE_MODIFIER_250DEG
        return {'x': x, 'y': y, 'z': z}

# ✅ 상보 필터 적용
def complementary_filter(accel_data, gyro_data, prev_angle, alpha=0.98, dt=0.2):
    accel_roll = np.arctan2(accel_data['y'], accel_data['z'])
    accel_pitch = np.arctan2(-accel_data['x'], np.sqrt(accel_data['y']**2 + accel_data['z']**2))

    gyro_roll = gyro_data['x'] * dt
    gyro_pitch = gyro_data['y'] * dt
    gyro_yaw = gyro_data['z'] * dt

    accel_yaw = np.arctan2(accel_data['y'], accel_data['x'])

    roll = alpha * (prev_angle['roll'] + gyro_roll) + (1 - alpha) * accel_roll
    pitch = alpha * (prev_angle['pitch'] + gyro_pitch) + (1 - alpha) * accel_pitch
    yaw = alpha * (prev_angle['yaw'] + gyro_yaw) + (1 - alpha) * accel_yaw

    return roll, pitch, yaw

def calibrate_imu(mpu, num_samples=1000):
    print("Calibrating IMU... Hold IMU still!")

    accel_bias = {'x': 0, 'y': 0, 'z': 0}
    gyro_bias = {'x': 0, 'y': 0, 'z': 0}

    for _ in range(num_samples):
        accel_data = mpu.get_accel_data()
        gyro_data = mpu.get_gyro_data()

        for axis in ['x', 'y', 'z']:
            accel_bias[axis] += accel_data[axis]
            gyro_bias[axis] += gyro_data[axis]

        time.sleep(0.01)

    for axis in ['x', 'y', 'z']:
        accel_bias[axis] /= num_samples
        gyro_bias[axis] /= num_samples

    accel_bias['z'] -= 9.80665  

    print(f"Calibration complete!\n"
          f"Accel Bias: {accel_bias}\n"
          f"Gyro Bias: {gyro_bias}")

    return accel_bias, gyro_bias

        # ✅ 데이터 스케일 다운 (10으로 나누기)
        for axis in ['x', 'y', 'z']:
            accel_data[axis] = (accel_data[axis] - accel_bias[axis])
            gyro_data[axis] = (gyro_data[axis] - gyro_bias[axis]) 

        # ✅ 상보 필터 적용
        roll, pitch, yaw = complementary_filter(
            accel_data, gyro_data, {'roll': roll, 'pitch': pitch, 'yaw': yaw}, alpha=0.996, dt=0.02
        )

        # ✅ Yaw 복구 로직 추가 (정지 시 보정)

        self.odom_pub = rospy.Publisher('/odom', Odometry, queue_size=10)

        # 센서 데이터 구독
        rospy.Subscriber('/scan', LaserScan, self.scan_callback)

        # 이전 Scan 데이터 저장
        self.prev_scan = None  
        self.last_time = rospy.Time.now()

        rospy.loginfo("🚀 ScanToOdom 노드가 시작되었습니다!")

    def scan_callback(self, msg):
        """ Scan 데이터를 가공하여 Odometry 메시지 형식으로 변환하여 퍼블리시 """
        current_time = rospy.Time.now()
        dt = (current_time - self.last_time).to_sec()
        self.last_time = current_time

        if self.prev_scan is None or dt <= 0:
            self.prev_scan = msg
            return

        # ICP를 수행하여 dx, dy, dtheta 계산
        dx, dy, dtheta = self.perform_scan_matching(self.prev_scan, msg)
        self.prev_scan = msg  # 이전 Scan 업데이트

        # 속도 계산
        vx = dx / dt if dt > 0 else 0.0
        vy = dy / dt if dt > 0 else 0.0
        omega = dtheta / dt if dt > 0 else 0.0

        # Odometry 메시지 생성
        odom_msg = Odometry()
        odom_msg.header.stamp = rospy.Time.now()
        odom_msg.header.frame_id = "odom"
        odom_msg.child_frame_id = "base_link"  # 로봇의 중심을 기준으로 설정

        # 위치 정보 (현재는 누적 X, EKF에서 보정 가능)
        odom_msg.pose.pose.position.x = 0.0
        odom_msg.pose.pose.position.y = 0.0

        # Orientation (Yaw 값 적용)
        quaternion = quaternion_from_euler(0, 0, dtheta)
        odom_msg.pose.pose.orientation = Quaternion(*quaternion)

        # Velocity 정보 (Scan Matching 기반으로 설정)
        odom_msg.twist.twist.linear.x = vx
        odom_msg.twist.twist.linear.y = vy
        odom_msg.twist.twist.angular.z = omega

        # 퍼블리시
        self.odom_pub.publish(odom_msg)

        rospy.loginfo(f"📌 Scan -> Odometry 변환 완료: vx={vx:.4f}, vy={vy:.4f}, omega={omega:.4f}")

    def perform_scan_matching(self, prev_scan, current_scan):
        """ Open3D의 ICP를 사용하여 Scan Matching 수행 """
        prev_points = self.scan_to_pointcloud(prev_scan)
        curr_points = self.scan_to_pointcloud(current_scan)

        if prev_points is None or curr_points is None:
            return 0.0, 0.0, 0.0  # Scan 데이터가 유효하지 않으면 0 반환

        # Open3D Point Cloud 객체 생성
        source = o3d.geometry.PointCloud()
        target = o3d.geometry.PointCloud()
        source.points = o3d.utility.Vector3dVector(curr_points)
        target.points = o3d.utility.Vector3dVector(prev_points)

        threshold = 0.05  # ICP 허용 오차
        trans_init = np.identity(4)  # 초기 변환 행렬

        # ICP 실행
        reg_p2p = o3d.pipelines.registration.registration_icp(
            source, target, threshold, trans_init,
            o3d.pipelines.registration.TransformationEstimationPointToPoint()
        )

        transformation = reg_p2p.transformation  # 최적화된 변환 행렬 (4x4)

        # 변환 행렬에서 이동량(dx, dy, dtheta) 추출
        dx, dy = transformation[0, 3], transformation[1, 3]
        dtheta = np.arctan2(transformation[1, 0], transformation[0, 0])  # 회전각 추출

        rospy.loginfo(f"📌 ICP 결과: dx={dx:.4f}, dy={dy:.4f}, dtheta={dtheta:.4f}")
        return dx, dy, dtheta
    
    def scan_to_pointcloud(self, scan_msg):
        """ LaserScan 데이터를 Point Cloud로 변환 """
        angles = np.linspace(scan_msg.angle_min, scan_msg.angle_max, len(scan_msg.ranges))
        x = np.array(scan_msg.ranges) * np.cos(angles)
        y = np.array(scan_msg.ranges) * np.sin(angles)
        points = np.vstack((x, y, np.zeros_like(x))).T

        return points if points.shape[0] > 0 else None

if __name__ == "__main__":
    try:
        scan_to_odom = ScanToOdom()
        rospy.spin()
    except rospy.ROSInterruptException:
        pass

INIT_V = 105
MIN_V = INIT_V - 35
MAX_V = INIT_V + 35

class PWMThrottleHat:
    def __init__(self, pwm, channel):
        self.pwm = pwm
        self.channel = channel
        self.pwm.frequency = 60  # 주파수 설정

    def set_throttle(self, throttle):
        pulse = int(0xFFFF * abs(throttle))  # 16비트 듀티 사이클 계산
        if throttle > 0:  # 전진
            self.pwm.channels[self.channel + 5].duty_cycle = pulse
            self.pwm.channels[self.channel + 4].duty_cycle = 0xFFFF
            self.pwm.channels[self.channel + 3].duty_cycle = 0
        elif throttle < 0:  # 후진
            self.pwm.channels[self.channel + 5].duty_cycle = pulse
            self.pwm.channels[self.channel + 4].duty_cycle = 0
            self.pwm.channels[self.channel + 3].duty_cycle = 0xFFFF
        else:  # 정지
            self.pwm.channels[self.channel + 5].duty_cycle = 0
            self.pwm.channels[self.channel + 4].duty_cycle = 0
            self.pwm.channels[self.channel + 3].duty_cycle = 0

class CmdVelSubscriber:
    def __init__(self):
        rospy.init_node('cmd_vel_subscriber', anonymous=True)
        rospy.Subscriber('/cmd_vel', Twist, self.listener_callback)

        # 모터 및 서보 컨트롤러 초기화
        i2c = busio.I2C(board.SCL, board.SDA)
        pca = PCA9685(i2c)
        pca.frequency = 60
        self.motor_hat = PWMThrottleHat(pca, channel=0)
        self.kit = ServoKit(channels=16, i2c=i2c, address=0x60)

        self.pan = INIT_V  # 초기 서보 위치
        self.kit.servo[0].angle = self.pan

        # 제어 변수
        self.linear_x = 0
        self.angular_z = 0
        self.running = True
        self.recovering = False

        rospy.on_shutdown(self.shutdown_callback)

    def shutdown_callback(self):
        """노드 종료 시 안전 정지"""
        rospy.loginfo("Shutting down the node.")
        self.running = False
        self.motor_hat.set_throttle(0)  # DC모터 정지
        self.kit.servo[0].angle = INIT_V  # 서보모터 초기화
        rospy.loginfo("Motor stopped and Servo initialized.")

    def listener_callback(self, msg):
        """cmd_vel 메시지 수신"""
        self.linear_x = msg.linear.x  # 전진/후진 속도
        self.angular_z = msg.angular.z  # 회전 속도
        rospy.loginfo(f"Received cmd_vel: linear_x={self.linear_x}, angular_z={self.angular_z}")

    def control_loop(self):
        """하나의 루프에서 리커버리, 조향 및 전진/후진을 수행"""
        dt = 0.1  # 업데이트 주기 (초)
        gain_factor = 200  # 회전 속도 조정 계수

        while self.running and not rospy.is_shutdown():
            # ✅ 1. 리커버리 모드 감지 및 처리 (제자리 회전)
            if abs(self.linear_x) <= 0.01 and self.angular_z != 0.0:
                self.recovering = True
                if self.angular_z > 0:
                    rospy.loginfo("Left Recovery: 제자리 회전")
                    self.kit.servo[0].angle = MAX_V
                    self.motor_hat.set_throttle(-1.0)
                    rospy.sleep(1)
                    self.kit.servo[0].angle = MIN_V
                    self.motor_hat.set_throttle(1.0)
                    rospy.sleep(1)

                elif self.angular_z < 0:
                    rospy.loginfo("Right Recovery: 제자리 회전")
                    self.kit.servo[0].angle = MIN_V
                    self.motor_hat.set_throttle(-1.0)
                    rospy.sleep(1)
                    self.kit.servo[0].angle = MAX_V
                    self.motor_hat.set_throttle(1.0)
                    rospy.sleep(1)

                self.recovering = False  # 리커버리 종료
                continue  # 리커버리 후 다시 루프 시작

            # ✅ 2. 조향 설정 (서보 모터) - 전진 시에만 조향 적용
            if self.linear_x > 0:
                if abs(self.angular_z) < 0.05:
                    self.pan = INIT_V  # 거의 회전하지 않으면 직진
                else:
                    # 전진 시에는 angular_z 반전하여 적용
                    effective_angular_z = -self.angular_z
                    angular_z_deg = math.degrees(effective_angular_z)
                    self.pan += dt * angular_z_deg * gain_factor
                    self.pan = max(min(self.pan, MAX_V), MIN_V)  # 서보 각도 제한
            else:
                if abs(self.angular_z) < 0.05:
                    self.pan = INIT_V  # 거의 회전하지 않으면 직진
                else:
                    #self.pan = INIT_V
                    effective_angular_z = self.angular_z
                    angular_z_deg = math.degrees(effective_angular_z)
                    self.pan += dt * angular_z_deg * gain_factor
                    self.pan = max(min(self.pan, MAX_V), MIN_V)  # 서보 각도 제한

            # 서보 모터 적용
            self.kit.servo[0].angle = self.pan
            rospy.loginfo(f"Servo angle updated: {self.pan:.2f} degrees")

            # ✅ 3. 전진/후진 제어 (DC 모터)
            if self.linear_x > 0:
                self.motor_hat.set_throttle(1)  # 전진
            elif self.linear_x < 0:
                self.motor_hat.set_throttle(-1)  # 후진
            else:
                self.motor_hat.set_throttle(0)  # 정지
            rospy.loginfo(f"Throttle set to: {self.linear_x}")

            #rospy.sleep(dt)


    def run(self):
        """제어 루프 실행"""
        self.control_loop()

            with open(CONFIG_PATH, "r") as file:
                return json.load(file)

            rospy.logerr(f"Config 파일 로드 실패: {e}")
            sys.exit(1)

            while not rospy.is_shutdown():
                print("Hold a tag near the reader")
                id, text = self.reader.read()  # 태그 읽기
                book_id = str(id)  # ID 값을 문자열로 변환
                rospy.loginfo(f"Tag ID: {book_id}, Text: {text}")

        self.IN3 = 16   # 모터 B IN3
        self.IN4 = 18   # 모터 B IN4

        self.SPEED_DELAY = speed_delay  # 속도 조절

        # 📍 GPIO 초기화
        GPIO.setmode(GPIO.BOARD)
        GPIO.setup(self.IN1, GPIO.OUT)
        GPIO.setup(self.IN2, GPIO.OUT)
        GPIO.setup(self.IN3, GPIO.OUT)
        GPIO.setup(self.IN4, GPIO.OUT)

        # 📍 RFID 리더 초기화
        self.reader = SimpleMFRC522()

        # ROS 노드 초기화
        rospy.init_node('rfid_motor_node', anonymous=True)

    def motor_forward(self):
        """모터를 전진시키는 함수"""
        rospy.loginfo("🚗 모터: FORWARD (이동 중)")
        for _ in range(5):  
            GPIO.output(self.IN1, GPIO.HIGH)
            GPIO.output(self.IN2, GPIO.LOW)
            GPIO.output(self.IN3, GPIO.HIGH) 
            GPIO.output(self.IN4, GPIO.LOW)
            time.sleep(self.SPEED_DELAY)

            GPIO.output(self.IN1, GPIO.LOW)
            GPIO.output(self.IN2, GPIO.LOW)
            GPIO.output(self.IN3, GPIO.LOW)
            GPIO.output(self.IN4, GPIO.LOW)
            time.sleep(self.SPEED_DELAY)

    def stop_motor(self):
        """모터를 정지시키는 함수"""
        GPIO.output(self.IN1, GPIO.LOW)
        GPIO.output(self.IN2, GPIO.LOW)
        GPIO.output(self.IN3, GPIO.LOW)
        GPIO.output(self.IN4, GPIO.LOW)
        rospy.loginfo("🛑 모터: STOPPED (정지)")

    def rfid_motor_control(self):
        """RFID 태깅을 통해 모터 제어"""
        rospy.loginfo("🔹 모터 시작: 첫 번째 RFID를 찾는 중...")
        self.motor_forward()  # 프로그램 시작 시 모터 바로 작동

        while not rospy.is_shutdown():
            rospy.loginfo("🔹 첫 번째 RFID를 태그하세요...")
            id1, text1 = self.reader.read()
            rospy.loginfo(f"✅ 첫 번째 RFID 인식됨: {id1}, {text1.strip()}")

            start_time = time.time()
            rfid_detected = False

            while time.time() - start_time < 10:
                self.motor_forward()
                rospy.loginfo("⏳ 두 번째 RFID를 기다리는 중...")
                id2, text2 = self.reader.read_no_block()
                
                if id2:
                    rospy.loginfo(f"✅ 두 번째 RFID 인식됨: {id2}, {text2.strip()}")
                    rfid_detected = True
                    break

            self.stop_motor()

            if not rfid_detected:
                # 두번쨰로 갈곳이 없으면  0 있으면 1 
                rospy.loginfo("⏰ 10초 동안 두 번째 RFID 인식되지 않아 자동 정지.")
                return 0
            else : 
                return 1

        self.cleanup()

    def cleanup(self):
        """GPIO 리소스를 정리하는 함수"""
        self.stop_motor()
        GPIO.cleanup()
        rospy.loginfo("🧹 GPIO 정리 완료!")

if __name__ == '__main__':
    try:
        controller = RFIDMotorController()
        controller.rfid_motor_control()
    except rospy.ROSInterruptException:
        controller.cleanup()
